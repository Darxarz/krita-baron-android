cmake_minimum_required(VERSION 3.16)
include(native/BaronVersion.cmake)
baron_read_version("${CMAKE_CURRENT_SOURCE_DIR}/android/version.json")
project(KritaBaronNative VERSION ${BARON_VERSION} LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
set(CMAKE_AUTORCC ON)
if(MSVC)
  add_compile_options(/utf-8)
endif()
find_package(Qt5 5.15 REQUIRED COMPONENTS Core Gui Widgets Network WebSockets Concurrent Qml Test)
add_library(baron_native STATIC native/OrchestrionClient.cpp native/JobQueue.cpp native/HistoryStore.cpp native/PluginUi.cpp native/plugin-icons.qrc native/BaronPanel.cpp native/GuidancePanel.cpp native/BaronUpdates.cpp native/ModelThumbnails.cpp native/Credentials.cpp native/Localization.cpp native/translations.qrc)
target_include_directories(baron_native PUBLIC native)
target_sources(baron_native PRIVATE native/BaronDiagnostics.cpp)
target_compile_definitions(baron_native PRIVATE BARON_VERSION="${BARON_VERSION}" BARON_VERSION_CODE=${BARON_VERSION_CODE})
target_sources(baron_native PRIVATE native/PersonalFonts.cpp native/ModelCatalog.cpp native/TagModel.cpp native/tags.qrc)
target_sources(baron_native PRIVATE native/InterfaceSettings.cpp)
target_sources(baron_native PRIVATE native/OrientationLayouts.cpp native/InpaintWidget.cpp)
target_sources(baron_native PRIVATE native/AuthorCredits.cpp)
target_sources(baron_native PRIVATE native/CloudWorkflow.cpp)
target_sources(baron_native PRIVATE native/UpscaleWidget.cpp)
target_sources(baron_native PRIVATE native/ConnectionButton.cpp native/PromptLogic.cpp native/PromptActions.cpp native/prompt.qrc)
target_compile_definitions(baron_native PUBLIC QT_USE_QSTRINGBUILDER QT_USE_FAST_CONCATENATION QT_USE_FAST_OPERATOR_PLUS)
target_link_libraries(baron_native PUBLIC Qt5::Widgets Qt5::Network Qt5::WebSockets Qt5::Concurrent Qt5::Qml)
set_target_properties(baron_native PROPERTIES POSITION_INDEPENDENT_CODE ON)
if(ANDROID)
  find_package(Qt5 REQUIRED COMPONENTS AndroidExtras)
  target_link_libraries(baron_native PRIVATE Qt5::AndroidExtras)
endif()
option(BARON_BUILD_PREVIEW "Build standalone UI preview (not Krita)" ON)
if(BARON_BUILD_PREVIEW)
  add_executable(baron_preview native/preview.cpp)
  target_link_libraries(baron_preview PRIVATE baron_native)
endif()
include(CTest)
if(BUILD_TESTING AND NOT ANDROID)
  add_executable(baron_tests tests/ClientTest.cpp)
  target_link_libraries(baron_tests PRIVATE baron_native Qt5::Test)
  add_test(NAME baron_native COMMAND baron_tests)
  set_tests_properties(baron_native PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
endif()
