cmake_minimum_required(VERSION 3.16)
project(KritaBaronNative VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
set(CMAKE_AUTORCC ON)
if(MSVC)
  add_compile_options(/utf-8)
endif()
find_package(Qt5 5.15 REQUIRED COMPONENTS Core Gui Widgets Network Test)
add_library(baron_native STATIC native/OrchestrionClient.cpp native/BaronPanel.cpp native/Credentials.cpp native/Localization.cpp native/translations.qrc)
target_include_directories(baron_native PUBLIC native)
target_link_libraries(baron_native PUBLIC Qt5::Widgets Qt5::Network)
set_target_properties(baron_native PROPERTIES POSITION_INDEPENDENT_CODE ON)
if(ANDROID)
  find_package(Qt5 REQUIRED COMPONENTS AndroidExtras)
  target_link_libraries(baron_native PRIVATE Qt5::AndroidExtras)
endif()
option(BARON_BUILD_PREVIEW "Build standalone UI preview (not Krita)" ON)
if(BARON_BUILD_PREVIEW)
  add_executable(baron_preview native/preview.cpp)
  target_link_libraries(baron_preview PRIVATE baron_native)
endif()
include(CTest)
if(BUILD_TESTING AND NOT ANDROID)
  add_executable(baron_tests tests/ClientTest.cpp)
  target_link_libraries(baron_tests PRIVATE baron_native Qt5::Test)
  add_test(NAME baron_native COMMAND baron_tests)
  set_tests_properties(baron_native PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
endif()
